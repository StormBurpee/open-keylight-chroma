cmake_minimum_required(VERSION 3.20)
project(open_keylight_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME update COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/update/run_tests.py)
add_test(NAME controller_http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/controller_update/run_tests.py)
add_test(NAME controller_job COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/controller_job/run_tests.py)
add_test(NAME services COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/services/run_tests.py)
add_test(NAME worker COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/worker/run_tests.py)
add_test(NAME transport COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/transport/run_tests.py)
add_test(NAME loader COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/loader/run_tests.py)
add_test(NAME controller_worker COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/controller_worker/run_tests.py)
add_executable(test_policy tests/test_policy.c firmware/components/keylight_json/keylight_policy.c)
target_include_directories(test_policy PRIVATE firmware/components/keylight_json/include)
add_test(NAME policy COMMAND test_policy)
add_executable(test_output tests/test_output.c firmware/main/output_policy.c
    firmware/components/keylight_core/keylight_core.c firmware/components/keylight_nxp/okl_nxp.c)
target_include_directories(test_output PRIVATE firmware/main firmware/components/keylight_core/include
    firmware/components/keylight_nxp/include)
if(NOT MSVC)
  target_compile_options(test_output PRIVATE -Wall -Wextra -Werror -Wpedantic)
  target_link_libraries(test_output m)
endif()
add_test(NAME output COMMAND test_output)
add_executable(test_core tests/test_core.c firmware/components/keylight_core/keylight_core.c)
target_include_directories(test_core PRIVATE firmware/components/keylight_core/include)
if(NOT MSVC)
  target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -Wpedantic)
  target_link_libraries(test_core m)
endif()
add_subdirectory(firmware-nxp)
add_test(NAME core COMMAND test_core)
add_executable(test_driver tests/test_driver.c firmware/components/keylight_nxp/okl_nxp.c firmware/components/keylight_nxp/okl_button.c)
target_include_directories(test_driver PRIVATE firmware/components/keylight_nxp/include)
if(NOT MSVC)
  target_compile_options(test_driver PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
add_test(NAME driver COMMAND test_driver)
find_path(CJSON_INCLUDE_DIR cJSON.h PATH_SUFFIXES cjson)
find_library(CJSON_LIBRARY cjson)
if(CJSON_INCLUDE_DIR AND CJSON_LIBRARY)
  add_executable(test_json tests/test_json.c firmware/components/keylight_json/keylight_json.c firmware/components/keylight_core/keylight_core.c)
  target_include_directories(test_json PRIVATE firmware/components/keylight_core/include firmware/components/keylight_json/include ${CJSON_INCLUDE_DIR})
  target_link_libraries(test_json ${CJSON_LIBRARY})
  if(NOT MSVC)
    target_compile_options(test_json PRIVATE -Wall -Wextra -Werror -Wpedantic)
    target_link_libraries(test_json m)
  endif()
  add_test(NAME json COMMAND test_json)
else()
  message(FATAL_ERROR "cJSON development library required for API validation tests")
endif()
